    function esc(value) {

      return String(value ?? '')

        .replaceAll('&', '&amp;')

        .replaceAll('<', '&lt;')

        .replaceAll('>', '&gt;')

        .replaceAll('"', '&quot;')

        .replaceAll("'", '&#039;');

    }


    function setContent(html) {

      document.getElementById('content').innerHTML = html;

    function showError(err) {

      setContent(`

          <div class="font-bold">Error</div>

          <div>${esc(err.message || err)}</div>

        </div>

      `);

    }


    async function apiGet(resource, params = {}) {

      const qs = new URLSearchParams({ resource, ...params });

      const res = await fetch(`${api}?${qs.toString()}`);

      const data = await res.json();

      if (!res.ok) throw new Error(data.error || `Request failed with status ${res.status}`);

      return data;

    }


    async function apiPost(action, body = {}) {

      const res = await fetch(api, {

        method: 'POST',

        headers: { 'Content-Type': 'application/json' },

        body: JSON.stringify({ action, ...body })

      });

      const data = await res.json();

      if (!res.ok) throw new Error(data.error || `Request failed with status ${res.status}`);

      return data;

    }


    function renderNav() {

      const html = tabs.map(([key, label]) => {

        const active = key === currentTab;

        return `

          >

            ${esc(label)}

          </button>

        `;

      }).join('');

      document.getElementById('nav').innerHTML = html;

    }


    function setTab(tab) {

      currentTab = tab;

      renderNav();

      loadData();


      if (timer) clearInterval(timer);

      const interval = tab === 'packing' ? 15000 : 60000;

      timer = setInterval(loadData, interval);

    }


    async function loadData() {

      try {

        document.getElementById('last-updated').textContent =

          `Updated ${new Date().toLocaleTimeString()}`;


        if (currentTab === 'dashboard') {

          const data = await apiGet('dashboard');

          renderDashboard(data);

        }


        if (currentTab === 'packing') {

          const data = await apiGet('packing-queue');

          renderPacking(data);

        }


        if (currentTab === 'returns') {

          const data = await apiGet('returns-pending');

          renderReturns(data);

        }


        if (currentTab === 'inventory') {

          const [inventory, skus] = await Promise.all([

            apiGet('inventory'),

            apiGet('sku-master')

          ]);

          renderInventory(inventory, skus);

        }


        if (currentTab === 'manufacturing') {

          const data = await apiGet('manufacturing-plan', mfgParams);

          renderManufacturing(data);

        }


        if (currentTab === 'sku') {

      }

    }


    function card(label, value, sub = '') {

      return `

        <div class="bg-white rounded shadow p-4">

          <div class="text-sm text-slate-500">${esc(label)}</div>

          <div class="text-2xl font-bold">${esc(value ?? 0)}</div>

          ${sub ? `<div class="text-xs text-slate-400 mt-1">${esc(sub)}</div>` : ''}

        </div>

      `;

    }


    function table(rows, columns, emptyMessage = 'No records found') {

      if (!rows || !rows.length) {

        return `<div class="bg-white rounded shadow p-4 text-slate-500">${esc(emptyMessage)}</div>`;

      }


      const head = columns.map(col => `<th class="text-left p-3 border-b">${esc(col.label)}</th>`).join('');

      const body = rows.map(row => {

        const cells = columns.map(col => {

          const value = col.render ? col.render(row) : row[col.key];

          return `<td class="p-3 border-b align-top">${value}</td>`;

        }).join('');

        return `<tr>${cells}</tr>`;

      }).join('');


      return `

        <div class="overflow-auto bg-white rounded shadow">

          <table class="min-w-full text-sm">

            <thead class="bg-slate-50">

              <tr>${head}</tr>

            </thead>

            <tbody>${body}</tbody>

          </table>

        </div>

      `;

    }


    function normalizeSummary(input) {

      if (!input) return {};

      if (Array.isArray(input)) input = input[0];

      if (input && input.dashboard_summary) return input.dashboard_summary;

      return input;

    }

    async function syncNow() {

      const btn = document.getElementById('sync-button');

      btn.disabled = true;

      btn.textContent = 'Syncing...';


      try {

        const res = await fetch('/.netlify/functions/sync?manual=1');

        const data = await res.json();

        if (!res.ok) throw new Error(data.error || 'Sync failed');

        alert(`Sync completed.\nOrders imported: ${data.importedOrders || 0}\nReturns imported: ${data.importedReturns || 0}`);

        await loadData();

      } catch (err) {

        alert(err.message || 'Sync failed');

      } finally {

        btn.disabled = false;

        btn.textContent = 'Sync now';

      }

    }


    async function onClickAction(e) {

      const btn = e.target.closest('button[data-action]');

      if (!btn) return;


      const action = btn.dataset.action;


      try {

        if (action === 'dispatch') {

          await apiPost('mark-dispatched', { order_db_id: Number(btn.dataset.id) });

          await loadData();

        }


        if (action === 'classify') {

          await apiPost('classify-return', {

            return_db_id: Number(btn.dataset.id),

            classification: btn.dataset.classification

          });

          await loadData();

        }


        if (action === 'map') {

          const master = prompt('Enter Master SKU ID to map to:');

          if (!master) return;


          await apiPost('map-sku', {

            platform: btn.dataset.platform,

            platform_sku_id: btn.dataset.platformSku,

            master_sku_id: master.trim()

          });


          await loadData();

        }

      } catch (err) {

        alert(err.message || 'Action failed');

      }

    }


    async function onFormSubmit(e) {

      const form = e.target;

      if (!form.dataset.form) return;


      e.preventDefault();


      const fd = new FormData(form);


      try {

        if (form.dataset.form === 'add-inventory') {

          await apiPost('add-inventory', {

            master_sku_id: fd.get('master_sku_id'),

            quantity: Number(fd.get('quantity') || 0),

            note: fd.get('note') || ''

          });

          alert('Inventory saved.');

          await loadData();

        }


        if (form.dataset.form === 'process-to-sellable') {

          await apiPost('process-to-sellable', {

            master_sku_id: fd.get('master_sku_id'),

            quantity: Number(fd.get('quantity') || 0)

          });

          alert('Processing stock moved to sellable.');

          await loadData();

        }


        if (form.dataset.form === 'add-manufacturing') {

          await apiPost('add-manufacturing', {

            master_sku_id: fd.get('master_sku_id'),

            quantity: Number(fd.get('quantity') || 0)

          });

          alert('Manufactured stock added.');

          await loadData();

        }


        if (form.dataset.form === 'import-orders') {

          const parsed = JSON.parse(fd.get('orders_json') || '[]');

          const orders = Array.isArray(parsed) ? parsed : parsed.orders || [];

          const result = await apiPost('import-orders', { orders });

          alert(`Orders imported: ${result.imported || 0}`);

          await loadData();

        }


        if (form.dataset.form === 'import-returns') {

          const parsed = JSON.parse(fd.get('returns_json') || '[]');

          const returns = Array.isArray(parsed) ? parsed : parsed.returns || [];

          const result = await apiPost('import-returns', { returns });

          alert(`Returns imported: ${result.imported || 0}`);

          await loadData();

        }

      } catch (err) {

        alert(err.message || 'Save failed');

      }

    }


    function onContentChange(e) {

      const control = e.target.dataset?.control;

      if (!control) return;


      if (control === 'mfg-weeks') {

        mfgParams.weeks = Number(e.target.value);

        loadData();

      }


      if (control === 'mfg-method') {

        mfgParams.method = e.target.value;

        loadData();

      }

    }


    document.getElementById('nav').addEventListener('click', e => {

      const btn = e.target.closest('button[data-tab]');

      if (btn) setTab(btn.dataset.tab);

    });


    document.getElementById('content').addEventListener('click', onClickAction);

    document.getElementById('content').addEventListener('submit', onFormSubmit);

    document.getElementById('content').addEventListener('change', onContentChange);

    document.getElementById('sync-button').addEventListener('click', syncNow);


    renderNav();

    setTab('dashboard');
